# PR Helper Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `pr()` shell function that auto-commits a dirty worktree as `wip: <branch>`, pushes it, and creates or reuses the GitHub PR with an AI-generated title/body or a random fallback when AI is unavailable.

**Architecture:** Keep the logic inside `shell/functions.sh`, where the other shell helpers live. `shell/init.sh` already sources that file, so no bootstrapping changes are needed. The function stays branch-local, uses only `git`, `gh`, `pi`, and `jq`, and returns early on detached HEAD or missing `gh`.

**Tech Stack:** Bash, Git, GitHub CLI (`gh`), OpenCode/`pi`, `jq`.

## Global Constraints

- Operate on the current git worktree only.
- Use the current branch name from `git`.
- Fail fast if not inside a git repo or if HEAD is detached.
- If the worktree has changes, stage everything and create a `wip: <branch>` commit before pushing.
- Push the current branch to `origin` on every invocation.
- After the push, if an open PR already exists for that branch, print its URL and exit.
- If no open PR exists, generate a PR title/body from the branch diff:
  - Try `pi` with the GPT-5.4 medium path and ask for strict JSON `{title, body}`.
  - Parse the JSON with `jq`.
  - If AI is unavailable or the output is invalid, synthesize a short fallback title/body with a random suffix so the PR can still move.
- Create the PR with explicit `gh pr create --title ... --body ...`.
- Keep the function self-contained; no new script file, no extra wrapper command, no new flags.
- Use `command git` / `command gh` / `command pi` so shell aliases do not interfere.
- Keep the logic branch-local: no branch switching, no worktree discovery, no cross-branch syncing.

```bash
pr() {
  command -v gh >/dev/null 2>&1 || {
    echo "pr: gh is not installed."
    return 1
  }

  command git rev-parse --is-inside-work-tree >/dev/null 2>&1 || {
    echo "pr: not inside a git repo."
    return 1
  }

  local branch pr_url
  branch="$(command git branch --show-current)"
  if [[ -z "$branch" ]]; then
    echo "pr: detached HEAD."
    return 1
  fi

  if [[ -n "$(command git status --porcelain --untracked-files=all)" ]]; then
    command git add -A || return 1
    if command git diff --cached --quiet --ignore-submodules --; then
      echo "pr: nothing to commit."
      return 1
    fi
    command git commit -m "wip: $branch" || return 1
  fi

  command git push -u origin HEAD || return 1

  pr_url="$(command gh pr list --head "$branch" --state open --json url --jq '.[0].url // empty')"
  if [[ -n "$pr_url" ]]; then
    printf '%s\n' "$pr_url"
    return 0
  fi

  local base_ref prompt ai_output title body fallback_suffix
  base_ref="$(command git symbolic-ref --quiet --short refs/remotes/origin/HEAD 2>/dev/null || printf '%s\n' origin/main)"
  prompt="$(cat <<EOF
Return strict JSON only with keys \"title\" and \"body\".
Write a concise GitHub pull request title and body for the changes below.
Keep the title short. Keep the body short, markdown-friendly, and useful.
No code fences. No extra prose.

Branch: $branch

Commit log:
$(command git log --oneline --no-merges --max-count=20 "$base_ref..HEAD")

Diff stat:
$(command git diff --stat "$base_ref...HEAD")
EOF
)"

  title=""
  body=""
  if command -v pi >/dev/null 2>&1; then
    ai_output="$(command pi --model opencode/gpt-5.4 "$prompt" 2>/dev/null || true)"
    if command -v jq >/dev/null 2>&1; then
      title="$(printf '%s' "$ai_output" | jq -r '.title // empty' 2>/dev/null)"
      body="$(printf '%s' "$ai_output" | jq -r '.body // empty' 2>/dev/null)"
    fi
  fi

  if [[ -z "$title" || -z "$body" ]]; then
    fallback_suffix="$(printf '%04x' "$RANDOM")"
    printf -v title 'wip: %s %s' "$branch" "$fallback_suffix"
    printf -v body 'Automated PR generated without AI.\n\nBranch: %s\nFallback: %s' "$branch" "$fallback_suffix"
  fi

  command gh pr create --title "$title" --body "$body"
}
```

tmpdir="$(mktemp -d)"
cat >"$tmpdir/git" <<'EOF'
#!/usr/bin/env bash
set -eo pipefail
case "$*" in
  "rev-parse --is-inside-work-tree")
    exit 0
    ;;
  "branch --show-current")
    printf '%s\n' feature/pr-helper
    ;;
  "status --porcelain --untracked-files=all")
    printf '%s\n' " M README.md"
    ;;
  "add -A")
    : "${ADD_CALLED:?}"
    : >"$ADD_CALLED"
    exit 0
    ;;
  "diff --cached --quiet --ignore-submodules --")
    exit 1
    ;;
  "commit -m wip: feature/pr-helper")
    : "${COMMIT_CALLED:?}"
    : >"$COMMIT_CALLED"
    exit 0
    ;;
  "push -u origin HEAD")
    : "${PUSH_CALLED:?}"
    : >"$PUSH_CALLED"
    exit 0
    ;;
  "symbolic-ref --quiet --short refs/remotes/origin/HEAD")
    printf '%s\n' origin/main
    ;;
  "log --oneline --no-merges --max-count=20 origin/main..HEAD")
    printf '%s\n' "abc1234 fix: add thing"
    ;;
  "diff --stat origin/main...HEAD")
    printf '%s\n' " README.md | 2 ++"
    ;;
  *)
    exit 0
    ;;
esac
EOF
cat >"$tmpdir/gh" <<'EOF'
#!/usr/bin/env bash
set -eo pipefail
case "$1 $2" in
  "pr list")
    printf '%s\n' "${PR_LIST_OUTPUT:-}"
    ;;
  "pr create")
    shift 2
    title=""
    body=""
    while [[ $# -gt 0 ]]; do
      case "$1" in
        --title)
          title="$2"
          shift 2
          ;;
        --body)
          body="$2"
          shift 2
          ;;
        *)
          shift
          ;;
      esac
    done
    [[ -n "$title" && -n "$body" ]] || exit 1
    : "${CREATE_CALLED:?}"
    printf '%s\n' "$title" >"$TITLE_FILE"
    printf '%s\n' "$body" >"$BODY_FILE"
    : >"$CREATE_CALLED"
    printf '%s\n' "created"
    exit 0
    ;;
  *)
    exit 0
    ;;
esac
EOF
cat >"$tmpdir/pi" <<'EOF'
#!/usr/bin/env bash
set -eo pipefail
: "${PI_CALLED:?}"
: >"$PI_CALLED"
case "${PI_MODE:-json}" in
  json)
    printf '%s\n' "${PI_JSON:-{\"title\":\"AI title\",\"body\":\"AI body\"}}"
    ;;
  fail)
    exit 127
    ;;
  *)
    printf '%s\n' "${PI_JSON:-{\"title\":\"AI title\",\"body\":\"AI body\"}}"
    ;;
esac
EOF
chmod +x "$tmpdir/git" "$tmpdir/gh" "$tmpdir/pi"
PATH="$tmpdir:$PATH"
export PATH

run_case() {
  rm -f "$tmpdir"/*.called "$tmpdir"/title.txt "$tmpdir"/body.txt 2>/dev/null || true
  ADD_CALLED="$tmpdir/add.called"
  COMMIT_CALLED="$tmpdir/commit.called"
  PUSH_CALLED="$tmpdir/push.called"
  CREATE_CALLED="$tmpdir/create.called"
  TITLE_FILE="$tmpdir/title.txt"
  BODY_FILE="$tmpdir/body.txt"
  PI_CALLED="$tmpdir/pi.called"
  export ADD_CALLED COMMIT_CALLED PUSH_CALLED CREATE_CALLED TITLE_FILE BODY_FILE PI_CALLED
}
